## 11.2 Separate the logic so it can be tested

Code: JavaScript:

```javascript
// Menu functions that don't touch the page, kept separate so they can be tested automatically

function filterMenu(items, category) {
  if (category === "All") {
    return items;
  }
  return items.filter((item) => item.category === category);
}

function sortItemsByPrice(items) {
  return [...items].sort((a, b) => a.price - b.price);
}

// Export for Node.js tests; the browser skips this block
if (typeof module !== "undefined") {
  module.exports = { filterMenu, sortItemsByPrice };
}
```

Code: JavaScript:

```javascript

  renderMenu(sortItemsByPrice(shownItems));
```

## 11.3 The test suite

Code: JavaScript:

```javascript
const test = require("node:test");
const assert = require("node:assert");
const { filterMenu, sortItemsByPrice } = require("../menu-logic.js");

const sample = [
  { name: "Latte", price: 4.5, category: "Coffee" },
  { name: "Thai Iced Tea", price: 2.5, category: "Tea" },
  { name: "Espresso", price: 3, category: "Coffee" },
  { name: "Brownie", price: 4.5, category: "Pastries" }
];

test("filtering Coffee returns only coffee items", () => {
  const result = filterMenu(sample, "Coffee");
  assert.deepStrictEqual(result.map((item) => item.name), ["Latte", "Espresso"]);
});
```

Command:

```bash
node --test
```

## 11.4 The workflow: tell GitHub to run the tests

File: YAML:

```yaml
name: Menu tests

on:
  pull_request:
  push:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - name: Check out the code
        uses: actions/checkout@v7

      - name: Set up Node.js
        uses: actions/setup-node@v7
        with:
          node-version: 24

      - name: Run the tests
        run: node --test
```
